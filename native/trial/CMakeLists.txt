 set(JUCE_PATH "" CACHE PATH "Pinned JUCE 8 source")
 add_subdirectory(${JUCE_PATH} JUCE)
 enable_testing()
 option(ZASU_TRIAL_SANITIZERS "Detect native trial memory lifetime errors in developer tests" OFF)
 if(ZASU_TRIAL_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address)
 endif()
 add_executable(ZasuTrialCoreTests TrialCoreTests.cpp)
 add_test(NAME ZasuTrialCoreTests COMMAND ZasuTrialCoreTests)
 add_executable(ZasuTrialRequestTests TrialRequestTests.cpp)
 find_package(Threads REQUIRED)
 target_link_libraries(ZasuTrialRequestTests PRIVATE Threads::Threads)
 add_test(NAME ZasuTrialRequestTests COMMAND ZasuTrialRequestTests)
 juce_add_console_app(ZasuTrialLicenseTests PRODUCT_NAME "ZASU Trial License Tests")
 target_sources(ZasuTrialLicenseTests PRIVATE TrialLicenseTests.cpp TrialLicense.cpp)
 target_compile_definitions(ZasuTrialLicenseTests PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
 target_link_libraries(ZasuTrialLicenseTests PRIVATE juce::juce_cryptography juce::juce_recommended_config_flags)
 juce_add_console_app(ZasuTrialControllerTests PRODUCT_NAME "ZASU Trial Controller Tests")
 target_sources(ZasuTrialControllerTests PRIVATE TrialControllerTests.cpp TrialController.cpp TrialLicense.cpp)
 target_compile_definitions(ZasuTrialControllerTests PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 ZASU_TRIAL_ACCOUNT_URL="http://127.0.0.1:8765/account/")
 target_link_libraries(ZasuTrialControllerTests PRIVATE juce::juce_cryptography juce::juce_recommended_config_flags)
 set(ZASU_TRIAL_TEST_FIXTURE "" CACHE FILEPATH "Ephemeral signed fixture for native controller tests")
 if(EXISTS "${ZASU_TRIAL_TEST_FIXTURE}")
  add_test(NAME ZasuTrialLicenseTests COMMAND ZasuTrialLicenseTests "${ZASU_TRIAL_TEST_FIXTURE}")
  add_test(NAME ZasuTrialControllerTests COMMAND ZasuTrialControllerTests "${ZASU_TRIAL_TEST_FIXTURE}")
  set_tests_properties(ZasuTrialControllerTests PROPERTIES TIMEOUT 60)
 endif()
 if(APPLE)
  target_link_libraries(ZasuTrialControllerTests PRIVATE "-framework Security")
 elseif(WIN32)
  target_link_libraries(ZasuTrialControllerTests PRIVATE bcrypt)
 endif()
endif()
function(zasu_enable_trial target product)
 set(ZASU_TRIAL_ACCOUNT_URL "https://zasuworks.jp/account/" CACHE STRING "Browser account page")
 if(NOT ZASU_TRIAL_ACCOUNT_URL STREQUAL "https://zasuworks.jp/account/")
  if(NOT ZASU_TRIAL_ACCOUNT_URL STREQUAL "http://127.0.0.1:8765/account/" OR NOT ZASU_TRIAL_ENDPOINT STREQUAL "https://giljyqaohutxoxlpkfjf.supabase.co/functions/v1/zasu-trial")
   message(FATAL_ERROR "Only the isolated staging API may use the fixed loopback validation page")
  endif()
 endif()
 target_sources(${target} PRIVATE ${CMAKE_CURRENT_FUNCTION_LIST_DIR}/TrialLicense.cpp ${CMAKE_CURRENT_FUNCTION_LIST_DIR}/TrialController.cpp)
 target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_FUNCTION_LIST_DIR})
 target_compile_definitions(${target} PRIVATE ZASU_TRIAL_BUILD=1 ZASU_TRIAL_ACCOUNT_URL="${ZASU_TRIAL_ACCOUNT_URL}" ZASU_TRIAL_PRODUCT="${product}" ZASU_TRIAL_RSA_MODULUS="${ZASU_TRIAL_RSA_MODULUS}" ZASU_TRIAL_ENDPOINT="${ZASU_TRIAL_ENDPOINT}")
 target_link_libraries(${target} PRIVATE juce::juce_cryptography)
 if(APPLE)
  target_link_libraries(${target} PRIVATE "-framework Security")
 elseif(WIN32)
  target_link_libraries(${target} PRIVATE bcrypt)
 endif()
endfunction()
